“Celdas de la sierra que ardieron 5 o más años desde 2000 y llevan 3 años sin fuego”
SELECT h3_cell, years_burned, last_fire_year
FROM cell_stats
WHERE area_id = 'INEGI:05004'
  AND years_burned >= 5
  AND last_fire_year <= 2023
ORDER BY years_burned DESC;
